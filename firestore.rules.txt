// Regras do Firestore para o projeto NOVO, só do site.
// Firestore > Regras: apague o que estiver lá, cole tudo isto e clique em Publicar.
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function ehEditorDoSite() {
      return request.auth != null
        && request.auth.token.email != null
        && exists(/databases/$(database)/documents/site-editores/$(request.auth.token.email.lower()));
    }

    // Conteúdo do site: qualquer visitante lê, só editores alteram
    match /site-publico/{doc} {
      allow read: if true;
      allow write: if ehEditorDoSite();
    }
    match /site-fotos/{id} {
      allow read: if true;
      allow write: if ehEditorDoSite();
    }
    match /site-fotos-full/{id} {
      allow read: if true;
      allow write: if ehEditorDoSite();
    }
    match /site-logos/{id} {
      allow read: if true;
      allow write: if ehEditorDoSite();
    }

    // Currículos: qualquer visitante pode ENVIAR (com limites); só editores leem e apagam
    match /site-curriculos/{id} {
      allow create: if request.resource.data.keys().hasOnly(['nome','telefone','email','cargo','texto','pdf','pdfNome','consentimento','lido','criadoEm'])
        && request.resource.data.nome is string && request.resource.data.nome.size() >= 3 && request.resource.data.nome.size() <= 110
        && request.resource.data.telefone is string && request.resource.data.telefone.size() <= 30
        && request.resource.data.email is string && request.resource.data.email.size() <= 150
        && request.resource.data.cargo is string && request.resource.data.cargo.size() <= 70
        && request.resource.data.texto is string && request.resource.data.texto.size() <= 19000
        && request.resource.data.pdf is string && request.resource.data.pdf.size() <= 960000
        && request.resource.data.pdfNome is string && request.resource.data.pdfNome.size() <= 180
        && (request.resource.data.texto.size() >= 30 || request.resource.data.pdf.size() > 0)
        && request.resource.data.consentimento == true
        && request.resource.data.lido == false
        && request.resource.data.criadoEm == request.time;
      allow read, update, delete: if ehEditorDoSite();
    }

    // Lista de editores: cada pessoa só vê o próprio registro.
    // Inclusões e exclusões são feitas pelo console do Firebase.
    // Depoimentos: qualquer visitante pode ENVIAR (aguardando aprovação);
    // o público só lê os aprovados; editores aprovam, ocultam e apagam
    match /site-depoimentos/{id} {
      allow create: if request.resource.data.keys().hasOnly(['nome','vinculo','texto','consentimento','aprovado','criadoEm'])
        && request.resource.data.nome is string && request.resource.data.nome.size() >= 2 && request.resource.data.nome.size() <= 60
        && request.resource.data.vinculo is string && request.resource.data.vinculo.size() <= 60
        && request.resource.data.texto is string && request.resource.data.texto.size() >= 10 && request.resource.data.texto.size() <= 600
        && request.resource.data.consentimento == true
        && request.resource.data.aprovado == false
        && request.resource.data.criadoEm == request.time;
      allow read: if resource.data.aprovado == true || ehEditorDoSite();
      allow update, delete: if ehEditorDoSite();
    }

    match /site-editores/{email} {
      allow read: if request.auth != null && request.auth.token.email.lower() == email;
      allow write: if false;
    }

    // Todo o resto fica bloqueado
    match /{documento=**} {
      allow read, write: if false;
    }
  }
}
